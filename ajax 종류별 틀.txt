// 에이잭스
$.ajax({
    url: `/community/${page}`,
    method: "GET", // 요청 메서드
    dataType: "json", // 서버에서 JSON 응답 기대
    success: function(result) {
        // 성공 시 처리
        console.log("데이터 로딩 성공:", result);
    },
    error: function(xhr, status, error) {
        // 실패 시 처리
        console.error("데이터 로딩 실패:", error, "상태 코드:", xhr.status);
    }
});
